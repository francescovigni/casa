# /ask Chat Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A ChatGPT-style `/ask` page answering questions about Francesco from ~40 curated Q&A pairs, matched client-side (Fuse.js pre-pass + in-browser MiniLM embeddings), with contact-CTA fallback.

**Architecture:** Curated entries live in `src/data/qa.ts`. A build-time Node script embeds question+aliases with quantized MiniLM (transformers.js) and writes a committed `public/qa-embeddings.json`; model + ONNX wasm files are self-hosted under `public/` (gitignored, fetched at build). A vanilla-TS Astro island does retrieval fully client-side. No server changes.

**Tech Stack:** Astro 5, Tailwind 3, TypeScript, `@huggingface/transformers` v3 (build script + browser), `fuse.js`, vitest.

**Spec:** `docs/superpowers/specs/2026-08-05-ask-chat-design.md`

## Global Constraints

- Vanilla-TS islands only — no UI framework (site convention).
- Page is EN-only; IT header links to the same `/ask/` page.
- Do NOT add `/ask/` to `localePairs` — middleware redirects paired EN routes for Italian IPs and would 302-loop/force-cookie this EN-only page. Unpaired routes are skipped (see `src/middleware.ts:26-27`).
- No runtime requests leave the browser: model from `/models/`, wasm from `/ort/`, embeddings from `/qa-embeddings.json`, all same-origin.
- `public/qa-embeddings.json` is committed; `public/models/` and `public/ort/` are gitignored and produced by `scripts/embed-qa.mjs` at build.
- Answers: plain text (typed out) + optional `links[]` rendered after typing. This realizes the spec's "markdown, links allowed" without an HTML-in-typewriter mess.
- Reuse existing utility classes: `wrap`, `kicker`, `display`, `lead`, colors `ink/muted/faint/line/canvas`.
- Respect `prefers-reduced-motion`: skip typewriter, render instantly.
- Commits: short imperative, `feat:`/`test:`/`chore:` prefixes as in git log.

// Pure retrieval logic for the /ask page. Kept framework-free so it unit-tests
// in Node and bundles into the browser island unchanged.
import Fuse from "fuse.js";
import type { QAEntry } from "../data/qa";

export const SIM_THRESHOLD = 0.6;
export const FUSE_CONFIDENT = 0.25;
export const FUSE_WEAK = 0.45;

export function bestEmbeddingMatch(
  query: number[],
  entries: { id: string; vectors: number[][] }[],
): { id: string; score: number } | null {
  let best: { id: string; score: number } | null = null;
  for (const e of entries) {
    for (const v of e.vectors) {
      let dot = 0;
      for (let i = 0; i < v.length; i++) dot += v[i] * query[i];
      if (!best || dot > best.score) best = { id: e.id, score: dot };
    }
  }
  return best;
}

export function makeFuse(entries: QAEntry[]): Fuse<QAEntry> {
  return new Fuse(entries, {
    keys: ["question", "aliases"],
    includeScore: true,
    ignoreLocation: true,
    threshold: 0.6,
  });
}

export function fuseMatch(
  fuse: Fuse<QAEntry>,
  query: string,
  maxScore: number,
): { id: string; score: number } | null {
  const hit = fuse.search(query)[0];
  if (!hit || hit.score === undefined || hit.score > maxScore) return null;
  return { id: hit.item.id, score: hit.score };
}
```

// Build-time: self-host the MiniLM model + ORT wasm under public/, then embed
// every qa question+alias into public/qa-embeddings.json. Idempotent; network
// only on first run (or after deleting public/models).
import { mkdirSync, writeFileSync, readFileSync, existsSync, copyFileSync, readdirSync } from "node:fs";
import { createHash } from "node:crypto";
import { dirname, join } from "node:path";

const MODEL_ID = "Xenova/all-MiniLM-L6-v2";
const MODEL_DIR = `public/models/${MODEL_ID}`;
const FILES = ["config.json", "tokenizer.json", "tokenizer_config.json", "onnx/model_quantized.onnx"];

for (const f of FILES) {
  const dest = join(MODEL_DIR, f);
  if (existsSync(dest)) continue;
  const url = `https://huggingface.co/${MODEL_ID}/resolve/main/${f}`;
  console.log("fetch", url);
  const res = await fetch(url);
  if (!res.ok) throw new Error(`${url}: ${res.status}`);
  mkdirSync(dirname(dest), { recursive: true });
  writeFileSync(dest, Buffer.from(await res.arrayBuffer()));
}

// Self-host onnxruntime-web wasm so the browser never hits a CDN.
const ortDist = "node_modules/@huggingface/transformers/node_modules/onnxruntime-web/dist";
const ortSrc = existsSync(ortDist) ? ortDist : "node_modules/onnxruntime-web/dist";
mkdirSync("public/ort", { recursive: true });
for (const f of readdirSync(ortSrc).filter((f) => f.endsWith(".wasm") || f.endsWith(".mjs")))
  copyFileSync(join(ortSrc, f), join("public/ort", f));

const { pipeline, env } = await import("@huggingface/transformers");
env.localModelPath = "public/models";
env.allowRemoteModels = false;
const extractor = await pipeline("feature-extraction", MODEL_ID, { dtype: "q8" });

const { qa } = await import("../src/data/qa.ts");
const entries = [];
for (const e of qa) {
  const texts = [e.question, ...e.aliases];
  const out = await extractor(texts, { pooling: "mean", normalize: true });
  const [n, dim] = out.dims;
  const flat = Array.from(out.data);
  entries.push({
    id: e.id,
    vectors: Array.from({ length: n }, (_, i) =>
      flat.slice(i * dim, (i + 1) * dim).map((x) => +x.toFixed(6))),
  });
}

const qaHash = createHash("sha256").update(readFileSync("src/data/qa.ts")).digest("hex");
writeFileSync(
  "public/qa-embeddings.json",
  JSON.stringify({ modelId: MODEL_ID, dim: 384, qaHash, entries }),
);
console.log(`embedded ${entries.length} entries`);
```

        const extractor = await pipeline("feature-extraction", data.modelId, { dtype: "q8" });
        return {
          entries: data.entries,
          embed: async (t: string) => {
            const out = await extractor(t, { pooling: "mean", normalize: true });
            return Array.from(out.data as Float32Array);
          },
        };
      } catch (err) {
        console.warn("[ask] embedding engine unavailable, fuse-only", err);
        return null; // degraded mode: fuse with FUSE_WEAK ceiling
      }
    })();
    return enginePromise;
  }
  input.addEventListener("focus", () => void loadEngine(), { once: true });

  // ---- rendering ----
  function addUserMsg(text: string) {
    const el = document.createElement("div");
    el.className = "ml-auto max-w-[85%] rounded-2xl rounded-br-sm border border-line px-4 py-2 text-sm text-ink w-fit";
    el.textContent = text;
    thread.appendChild(el);
    el.scrollIntoView({ behavior: "smooth", block: "end" });
  }

  function addTyping(): HTMLElement {
    const el = document.createElement("div");
    el.className = "text-faint text-sm";
    el.textContent = "…";
    thread.appendChild(el);
    return el;
  }

  function renderAnswer(el: HTMLElement, text: string, links?: { label: string; href: string }[]) {
    el.className = "max-w-[85%] text-[15px] leading-relaxed text-muted";
    el.textContent = "";
    const done = () => {
      if (!links?.length) return;
      const row = document.createElement("p");
      row.className = "mt-3 flex gap-4";
      for (const l of links) {
        const a = document.createElement("a");
        a.href = l.href;
        a.textContent = l.label + " →";
        a.className = "link-underline text-sm font-medium text-ink";
        row.appendChild(a);
      }
      el.appendChild(row);
    };
    if (reduceMotion) {
      el.textContent = text;
      done();
      return;
    }
    let i = 0;
    const tick = () => {
      i = Math.min(i + 3, text.length);
      el.textContent = text.slice(0, i);
      if (i < text.length) requestAnimationFrame(tick);
      else done();
    };
    requestAnimationFrame(tick);
  }

  function answerById(id: string, typing: HTMLElement) {
    const e = byId.get(id)!;
    renderAnswer(typing, e.answer, e.links);
  }

  // ---- flows ----
  chipsBox.addEventListener("click", (ev) => {
    const btn = (ev.target as HTMLElement).closest<HTMLElement>("[data-chip-id]");
    if (!btn) return;
    const e = byId.get(btn.dataset.chipId!)!;
    addUserMsg(e.question);
    answerById(e.id, addTyping());
  });

  form.addEventListener("submit", async (ev) => {
    ev.preventDefault();
    const q = input.value.trim();
    if (!q) return;
    input.value = "";
    addUserMsg(q);
    const typing = addTyping();

    const confident = fuseMatch(fuse, q, FUSE_CONFIDENT);
    if (confident) return answerById(confident.id, typing);

    typing.textContent = "thinking…";
    const engine = await loadEngine();
    if (engine) {
      const m = bestEmbeddingMatch(await engine.embed(q), engine.entries);
      if (m && m.score >= SIM_THRESHOLD) return answerById(m.id, typing);
    } else {
      const weak = fuseMatch(fuse, q, FUSE_WEAK);
      if (weak) return answerById(weak.id, typing);
    }
    renderAnswer(typing, FALLBACK, FALLBACK_LINKS);
  });
</script>

```js
// Reports hit/miss accuracy of the embedding matcher over paraphrase probes
// at candidate thresholds. Run after editing qa.ts: node scripts/tune-threshold.mjs
const PROBES = [
  // [query, expected qa id or null for should-miss]
  ["experience with medical images and deep learning?", "healthcare-ai"],
  ["do you know kubernetes and helm", "infrastructure"],
  ["would you relocate for a job", "open-to-roles"],
  ["what's your favourite pizza topping", null],
  ["can you fix my printer", null],
  // extend as misses are discovered in real use
];

const { pipeline, env } = await import("@huggingface/transformers");
env.localModelPath = "public/models";
env.allowRemoteModels = false;
const { readFileSync } = await import("node:fs");
const data = JSON.parse(readFileSync("public/qa-embeddings.json", "utf8"));
const extractor = await pipeline("feature-extraction", data.modelId, { dtype: "q8" });
const { bestEmbeddingMatch } = await import("../src/lib/retrieval.ts");

for (const t of [0.5, 0.55, 0.6, 0.65, 0.7]) {
  let ok = 0;
  for (const [q, want] of PROBES) {
    const out = await extractor(q, { pooling: "mean", normalize: true });
    const m = bestEmbeddingMatch(Array.from(out.data), data.entries);
    const got = m && m.score >= t ? m.id : null;
    if (got === want) ok++;
  }
  console.log(`threshold ${t}: ${ok}/${PROBES.length}`);
}
```
